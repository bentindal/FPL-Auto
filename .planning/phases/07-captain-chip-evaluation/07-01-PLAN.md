def suggest_captaincy(self, strategy_config=None):
    """
    Suggest captain and vice-captain based on strategy mode.
    
    Args:
        strategy_config: Optional StrategyConfig with captain_mode, captain_lookback_gws, captain_variance_penalty.
                        If None, defaults to highest_xp (backward compat).
    
    Returns:
        ((captain_name, captain_xp), (vice_captain_name, vice_captain_xp))
    """
```

**Implementation (supporting all 3 modes):**

1. **Mode: highest_xp (default, current behavior)**
   - Get all squad players with _all_xp_dicts (multi-GW discounted xP)
   - Sort by xP descending
   - Return top 2 (captain, vice-captain)
   - Ignore lookback_gws and variance_penalty (they're 1 and 0 anyway)

2. **Mode: highest_value**
   - Get all squad players with their prices (use self.player_value(player, self.gw_data))
   - Sort by price descending
   - Return top 2 by price
   - Rationale: Highest-priced players are more stable (per CONTEXT.md)
   - Ignore lookback_gws and variance_penalty

3. **Mode: form_based**
   - Compute rolling form for each player using recent GW xP
   - Form = average of last captain_lookback_gws xP values from _all_xp_dicts
   - Apply variance penalty: score = form + (variance * captain_variance_penalty)
     - Variance = std deviation of player's xP over last captain_lookback_gws
     - variance_penalty=-0.2: subtract 0.2*variance (prefer high-variance picks)
   - Sort by adjusted score descending
   - Return top 2

**Error handling:**
- If squad has < 2 players, raise ValueError (existing behavior)
- If _all_xp_dicts is missing a player, use default xP of 0

**Backward compatibility:**
- If strategy_config is None, use highest_xp mode (existing behavior)
- auto_captain() will call with self.strategy_config (pass through)

class TestTeamCaptaincy(unittest.TestCase):
    """Test suggest_captaincy() with all 3 captain modes."""
    
    def setUp(self):
        """Create test team with known xP/price distribution."""
        # Create team with mock data (or use actual data from 2021-22 GW1)
        self.season = '2021-22'
        self.gw = 1
        self.team = Team(
            season=self.season,
            gameweek=self.gw,
            strategy_config=BASELINE_CURRENT  # Default
        )
    
    def test_suggest_captaincy_highest_xp(self):
        """Captain should be highest xP player."""
        config = CAPTAIN_HIGHEST_XP
        captain, vice = self.team.suggest_captaincy(strategy_config=config)
        # Verify captain[0] is highest xP in squad
        all_xp = self.team.get_all_xp()
        expected_captain = sorted(all_xp, key=lambda x: float(x[1]), reverse=True)[0]
        self.assertEqual(captain[0], expected_captain[0])
    
    def test_suggest_captaincy_form_based(self):
        """Captain should prefer recent hot streak with variance penalty."""
        config = CAPTAIN_FORM_BASED
        captain, vice = self.team.suggest_captaincy(strategy_config=config)
        # Verify captain xP > vice captain xP (should be top 2 by form score)
        self.assertGreaterEqual(float(captain[1]), float(vice[1]))
    
    def test_suggest_captaincy_highest_value(self):
        """Captain should be highest-priced player."""
        config = CAPTAIN_HIGHEST_VALUE
        captain, vice = self.team.suggest_captaincy(strategy_config=config)
        # Verify captain price >= vice price
        captain_price = self.team.player_value(captain[0], self.team.gw_data)
        vice_price = self.team.player_value(vice[0], self.team.gw_data)
        self.assertGreaterEqual(captain_price, vice_price)
    
    def test_auto_captain_with_strategy_config(self):
        """auto_captain() should use passed strategy_config."""
        config = CAPTAIN_FORM_BASED
        self.team.auto_captain(strategy_config=config)
        # Verify self.captain is set (not empty)
        self.assertNotEqual(self.team.captain, '')
    
    def test_auto_captain_without_strategy_config(self):
        """auto_captain() should use self.strategy_config if param not provided."""
        self.team.strategy_config = CAPTAIN_HIGHEST_VALUE
        self.team.auto_captain()
        # Verify captain is set
        self.assertNotEqual(self.team.captain, '')
    
    def test_suggest_captaincy_squad_size_check(self):
        """suggest_captaincy() should raise if squad < 2 players."""
        self.team.gks = []  # Empty GK list → squad < 2
        with self.assertRaises(ValueError):
            self.team.suggest_captaincy(strategy_config=CAPTAIN_HIGHEST_XP)
```
