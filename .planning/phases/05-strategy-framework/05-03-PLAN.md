```python
import numpy as np
from typing import Dict, List

def compute_season_metrics(weekly_points: List[float], baseline_weekly_points: List[float] = None) -> Dict:
    """
    Compute multi-dimensional metrics for a single season.
    
    Args:
        weekly_points: list of 38 GW point scores
        baseline_weekly_points: optional baseline for comparison
    
    Returns:
        dict with keys:
        - total_points: sum of season
        - mean_gw_points: average per week
        - std_gw_points: std dev per week
        - sharpe_ratio: (mean - rf) / std, rf=0 for FPL
        - sortino_ratio: (mean - rf) / std(downside)
        - coefficient_variation: std / mean (lower = more consistent)
        - max_drawdown: worst cumulative loss as % of peak
        - best_week: max single GW
        - worst_week: min single GW
        - win_rate: % weeks beating baseline (if baseline provided)
        - vs_baseline_points: total difference vs baseline (if baseline provided)
    """
    p = np.array(weekly_points)
    
    total = float(np.sum(p))
    mean = float(np.mean(p))
    std = float(np.std(p))
    
    # Sharpe: (mean - rf) / std, rf = 0
    sharpe = mean / std if std > 0 else 0.0
    
    # Sortino: (mean - rf) / std(downside)
    downside = np.std([x for x in p if x < mean]) if any(x < mean for x in p) else std
    sortino = mean / downside if downside > 0 else 0.0
    
    # Consistency: CV = std / mean
    cv = std / mean if mean > 0 else 0.0
    
    # Max drawdown: worst (peak - current) / peak
    cumsum = np.cumsum(p)
    peak = np.maximum.accumulate(cumsum)
    drawdown = (peak - cumsum) / peak
    max_dd = float(np.max(drawdown)) if len(drawdown) > 0 else 0.0
    
    metrics = {
        'total_points': total,
        'mean_gw_points': float(mean),
        'std_gw_points': float(std),
        'sharpe_ratio': float(sharpe),
        'sortino_ratio': float(sortino),
        'coefficient_variation': float(cv),
        'max_drawdown': float(max_dd),
        'best_week': float(np.max(p)),
        'worst_week': float(np.min(p)),
    }
    
    if baseline_weekly_points is not None:
        baseline = np.array(baseline_weekly_points)
        baseline_total = float(np.sum(baseline))
        win_rate = float(np.sum(p >= baseline) / len(p))
        
        metrics['vs_baseline_total_points'] = total - baseline_total
        metrics['vs_baseline_win_rate'] = win_rate
    
    return metrics
```

def bootstrap_ci(
    strategy_a_seasons: List[Dict],
    strategy_b_seasons: List[Dict] = None,
    metric_key: str = 'total_points',
    n_bootstrap: int = 10000,
    ci: float = 0.95,
) -> Dict:
    """
    Compute bootstrapped confidence intervals for metric comparison.
    
    Args:
        strategy_a_seasons: list of dicts with 'p_list' (weekly points) and metric_key
                           (e.g., [{'p_list': [...], 'total_points': 2145}, ...])
        strategy_b_seasons: optional second strategy for comparison
        metric_key: metric to compute CI for (e.g., 'sharpe_ratio', 'total_points')
        n_bootstrap: number of resampling iterations
        ci: confidence level (0.95 = 95%)
    
    Returns:
        dict with keys:
        - mean_a: mean of metric for strategy A
        - ci_lower_a: lower bound
        - ci_upper_a: upper bound
        - mean_b: mean of metric for strategy B (if provided)
        - ci_lower_b: lower bound
        - ci_upper_b: upper bound
        - diff_mean: A - B (if provided)
        - diff_ci_lower: lower CI on difference
        - diff_ci_upper: upper CI on difference
        - significant: whether CI excludes 0 (if B provided)
    """
    import numpy as np
    
    # Extract metric values for A
    a_values = np.array([s.get(metric_key, 0) for s in strategy_a_seasons])
    
    # Bootstrap for A
    a_diffs = []
    for _ in range(n_bootstrap):
        sample = np.random.choice(a_values, size=len(a_values), replace=True)
        a_diffs.append(np.mean(sample))
    
    a_diffs = np.array(a_diffs)
    alpha = 1 - ci
    a_mean = float(np.mean(a_values))
    a_lower = float(np.percentile(a_diffs, alpha / 2 * 100))
    a_upper = float(np.percentile(a_diffs, (1 - alpha / 2) * 100))
    
    result = {
        'mean_a': a_mean,
        'ci_lower_a': a_lower,
        'ci_upper_a': a_upper,
    }
    
    if strategy_b_seasons is not None:
        b_values = np.array([s.get(metric_key, 0) for s in strategy_b_seasons])
        
        # Bootstrap for difference
        diffs = []
        for _ in range(n_bootstrap):
            a_sample = np.random.choice(a_values, size=len(a_values), replace=True)
            b_sample = np.random.choice(b_values, size=len(b_values), replace=True)
            diffs.append(np.mean(a_sample) - np.mean(b_sample))
        
        diffs = np.array(diffs)
        b_mean = float(np.mean(b_values))
        b_lower = float(np.percentile(diffs, alpha / 2 * 100))
        b_upper = float(np.percentile(diffs, (1 - alpha / 2) * 100))
        
        result.update({
            'mean_b': b_mean,
            'ci_lower_b': b_lower,
            'ci_upper_b': b_upper,
            'diff_mean': a_mean - b_mean,
            'diff_ci_lower': float(np.percentile(diffs, alpha / 2 * 100)),
            'diff_ci_upper': float(np.percentile(diffs, (1 - alpha / 2) * 100)),
            'significant': 0.0 not in [float(np.percentile(diffs, alpha / 2 * 100)),
                                       float(np.percentile(diffs, (1 - alpha / 2) * 100))],
        })
    
    return result
```

def apply_bonferroni_correction(num_comparisons: int, alpha: float = 0.05) -> float:
    """
    Apply Bonferroni correction for multiple comparisons.
    
    Args:
        num_comparisons: number of statistical tests (e.g., 10 strategy pairs)
        alpha: base significance level (default 0.05)
    
    Returns:
        corrected_alpha: α / num_comparisons
    
    Example:
        If testing 10 strategy variants against baseline:
        corrected_alpha = apply_bonferroni_correction(10)  # 0.005
        Only reject null hypothesis if p-value < 0.005 (stricter)
    """
    return alpha / num_comparisons


def format_metrics_table(strategy_results: Dict) -> str:
    """
    Format multi-strategy results as markdown table.
    
    Args:
        strategy_results: dict {strategy_name: {
            'test_metrics': [...],
            'mean_metrics': {'sharpe_ratio': ..., 'total_points': ...},
            'ci_metrics': {...}
        }}
    
    Returns:
        markdown formatted table suitable for reporting
    """
    lines = [
        "| Strategy | Total Points | Mean ± 95% CI | Sharpe Ratio | Sortino Ratio | CV | Max Drawdown |",
        "|----------|--------------|---------------|--------------|---------------|----|----|",
    ]
    
    for name, data in strategy_results.items():
        mean_metrics = data.get('mean_metrics', {})
        ci_metrics = data.get('ci_metrics', {})
        
        total = mean_metrics.get('total_points', 0)
        sharpe = mean_metrics.get('sharpe_ratio', 0)
        sortino = mean_metrics.get('sortino_ratio', 0)
        cv = mean_metrics.get('coefficient_variation', 0)
        mdd = mean_metrics.get('max_drawdown', 0)
        
        # CI for total points
        total_ci_lower = ci_metrics.get('total_points', {}).get('ci_lower_a', 0)
        total_ci_upper = ci_metrics.get('total_points', {}).get('ci_upper_a', 0)
        
        line = f"| {name} | {total:.0f} | {total:.0f} [{total_ci_lower:.0f}, {total_ci_upper:.0f}] | {sharpe:.2f} | {sortino:.2f} | {cv:.2f} | {mdd:.1%} |"
        lines.append(line)
    
    return '\n'.join(lines)
```
