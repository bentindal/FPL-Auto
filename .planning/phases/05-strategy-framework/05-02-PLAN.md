def run_strategy_on_seasons(strategy_config, seasons):
    """
    Run a single strategy across multiple seasons and collect per-season results.
    
    Args:
        strategy_config: StrategyConfig instance (e.g., BASELINE_STATIC)
        seasons: list of season strings (e.g., ['2021-22', '2022-23'])
    
    Returns:
        list of dicts: [
            {
                'season': '2021-22',
                'p_list': [32, 44, 41, ...],  # 38 GW points
                'xp_list': [53.47, 42.32, ...],
                'chips_used': [...],
                'transfer_history': [...],
            },
            ...
        ]
    """
    import manager
    import multiprocessing
    
    configs = [
        {
            'season': season,
            'start_gw': 1,
            'repeat': 38 - 1,  # GW 1 to 38
            'starting_team': 'auto',
            'quiet': True,
            'strategy': strategy_config,
        }
        for season in seasons
    ]
    
    results = []
    with multiprocessing.Pool(processes=min(4, len(configs))) as pool:
        results = pool.map(manager.run_season, configs)
    
    return results
```

**Important:** This function imports manager and uses multiprocessing. It runs the strategy sequentially across seasons (not nested yet—nesting is in Task 2). Each season run returns raw p_list, xp_list (weekly points and xP).

Add docstring and type hints. Handle exceptions gracefully (season missing → return None, continue to next season).

def nested_walk_forward_evaluation(strategy_config, all_seasons=None):
    """
    Nested walk-forward validation: train on N-1 seasons, test on held-out season.
    
    Args:
        strategy_config: StrategyConfig instance to evaluate
        all_seasons: list of season strings. Default: ['2021-22', '2022-23', '2023-24', '2024-25']
    
    Returns:
        list of dicts: [
            {
                'iteration': 1,
                'test_season': '2023-24',
                'train_seasons': ['2021-22', '2022-23'],
                'test_results': {
                    'season': '2023-24',
                    'p_list': [...],
                    'xp_list': [...],
                    'total_points': 2145,
                    'avg_gw_points': 56.4,
                    ...
                },
                'train_metrics': {
                    'avg_total_points': 2187,
                    'avg_sharpe': 0.62,
                    ...
                }
            },
            ...
        ]
    """
    import numpy as np
    from datetime import datetime
    
    if all_seasons is None:
        all_seasons = ['2021-22', '2022-23', '2023-24', '2024-25']
    
    results = []
    
    # Outer loop: iterate over test seasons
    # For 4 seasons, test on last 2 (2023-24, 2024-25) for robustness
    for test_idx, test_season in enumerate(['2023-24', '2024-25']):
        train_seasons = [s for s in all_seasons if s != test_season]
        
        # Inner loop: run strategy on training seasons to collect baseline metrics
        train_results = run_strategy_on_seasons(strategy_config, train_seasons)
        train_metrics = aggregate_season_results(train_results)
        
        # Outer evaluation: run strategy on held-out test season
        test_results_raw = run_strategy_on_seasons(strategy_config, [test_season])
        test_results = test_results_raw[0] if test_results_raw else None
        
        if test_results is None:
            print(f"Warning: Could not run test season {test_season}")
            continue
        
        # Compute test metrics
        test_metrics = compute_season_metrics(
            weekly_points=test_results['p_list'],
            season=test_season
        )
        
        results.append({
            'iteration': test_idx + 1,
            'test_season': test_season,
            'train_seasons': train_seasons,
            'test_results': test_results,
            'test_metrics': test_metrics,
            'train_metrics': train_metrics,
            'timestamp': datetime.now().isoformat(),
        })
    
    return results
```

- aggregate_season_results(results): Takes list of per-season results, averages metrics across seasons

Keep metric computation minimal here (just total_points, avg, std). Full metrics (Sharpe, Sortino, CI) come in Plan 05-03.
