import json

# Load Phase 9 results and top 100 baseline
with open('evaluation/phase9_validation_results.json') as f:
  phase9 = json.load(f)

with open('evaluation/top100_baseline.json') as f:
  baseline = json.load(f)

our_mean = phase9['aggregate']['mean_total_points']
top100_mean = baseline['mean']
threshold_75pct = top100_mean * 0.75

# Percentile rank (assuming comparison to 2019-20 top 100)
percentile = (sum(1 for s in range(baseline['count']) if s < our_mean) / baseline['count'] * 100)

success = 'PASS' if our_mean >= threshold_75pct else 'FAIL'

# Generate markdown report
md = f'''# Phase 9: Percentile Ranking vs Top 100 Managers

## Comparison Summary

| Metric | Value | Notes |
|--------|-------|-------|
| **Our Strategy (PHASE_8_OPTIMAL)** | {our_mean:.0f} pts | Mean across {phase9['aggregate']['seasons_count']} test seasons |
| **Top 100 Average (2019-20)** | {top100_mean:.0f} pts | Historical baseline (single season) |
| **Success Threshold (≥75% of top 100)** | {threshold_75pct:.0f} pts | Pragmatic requirement per CONTEXT.md |
| **Percentile Rank** | {percentile:.1f}% | Our position vs top 100 distribution |
| **Success Criterion** | **{success}** | {our_mean:.0f} {'≥' if success == 'PASS' else '<'} {threshold_75pct:.0f} |

## Interpretation

- **Our Mean:** {our_mean:.0f} points across {phase9['aggregate']['seasons_count']} validation seasons
- **Top 100 Mean:** {top100_mean:.0f} points (2019-20 historical benchmark)
- **Difference:** {our_mean - top100_mean:+.0f} points ({(our_mean/top100_mean - 1)*100:+.1f}% vs baseline)
- **Percentile:** We rank at the {percentile:.0f}th percentile of historical top 100 managers

## Caveats

1. **Data Source:** Top 100 baseline is from 2019-20 season only (5+ years old). Current-season comparable data not available in repository.
2. **Time Frame:** Our strategy tested on 2023-24 + historical 2021-22, 2022-23 training. Different era, player pool, fixture distribution vs 2019-20.
3. **Methodology:** Walk-forward validation (no lookahead bias); top 100 managers may use different information access (real-time news, injury updates).

## Recommendation

**SUCCESS CRITERION MET:** {'YES ✓' if success == 'PASS' else 'NO ✗'}

Our strategy achieves {our_mean/threshold_75pct*100:.1f}% of the pragmatic 75% threshold.
'''

with open('evaluation/percentile_ranking.md', 'w') as f:
  f.write(md)

print(f'Percentile ranking: {success} at {percentile:.1f}th percentile')
"
