from fpl_auto.strategies import PHASE_8_OPTIMAL
from evaluation.walk_forward import nested_walk_forward_evaluation
from evaluation.metrics import compute_season_metrics, bootstrap_ci
import json

# Run walk-forward on all seasons (2024-25 skipped if data not fixed, per Plan 09-01)
seasons = ['2021-22', '2022-23', '2023-24']  # Adjust if 2024-25 available

results = nested_walk_forward_evaluation(PHASE_8_OPTIMAL, seasons)

# Extract per-season metrics
metrics_output = {
  'phase': '09-performance-validation',
  'strategy': 'PHASE_8_OPTIMAL',
  'seasons_tested': seasons,
  'per_season': [],
  'aggregate': {}
}

total_points_all = []
for iteration in results:
  season = iteration['test_season']
  total_pts = iteration['test_metrics']['total_points']
  total_points_all.append(total_pts)
  
  metrics_output['per_season'].append({
    'season': season,
    'total_points': total_pts,
    'sharpe_ratio': iteration['test_metrics'].get('sharpe_ratio'),
    'sortino_ratio': iteration['test_metrics'].get('sortino_ratio'),
    'max_drawdown': iteration['test_metrics'].get('max_drawdown'),
    'cv': iteration['test_metrics'].get('cv')
  })

# Aggregate metrics
ci_lower, ci_upper = bootstrap_ci(total_points_all, 0.95, 10000)
metrics_output['aggregate'] = {
  'mean_total_points': sum(total_points_all) / len(total_points_all),
  'total_points_95ci_lower': ci_lower,
  'total_points_95ci_upper': ci_upper,
  'seasons_count': len(total_points_all)
}

# Save results
with open('evaluation/phase9_validation_results.json', 'w') as f:
  json.dump(metrics_output, f, indent=2)

print('Phase 9 validation complete')
