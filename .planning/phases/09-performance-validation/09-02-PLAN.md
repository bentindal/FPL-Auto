```python
class TemporalAudit:
  def __init__(self, strategy_config, season):
    self.strategy_config = strategy_config
    self.season = season
    self.audit_log = []
  
  def audit_season(self):
    """Run single season with temporal instrumentation."""
    team = Team(...)
    for gw in range(1, 39):
      gate = TemporalGate(season=self.season, decision_gameweek=gw)
      
      # Decision 1: Transfer
      violations = []
      try:
        team.auto_transfer()
      except TemporalViolationError as e:
        violations.append(str(e))
      self.audit_log.append({'gw': gw, 'decision': 'transfer', 'violations': violations})
      
      # Decisions 2-4: captain, chips, subs (similar structure)
      ...
    
    return self._generate_verdicts()
  
  def _generate_verdicts(self):
    """Compute pass/fail for each decision point and overall."""
    verdicts = {}
    for decision in ['transfer', 'captain', 'chips', 'subs']:
      violations = [log for log in self.audit_log if log['decision'] == decision and log['violations']]
      verdicts[decision] = 'PASS' if not violations else 'FAIL'
    verdicts['overall'] = 'PASS' if all(v == 'PASS' for v in verdicts.values()) else 'FAIL'
    return verdicts
```
