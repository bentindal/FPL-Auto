```python
def avg_player_data(self, season, from_gw, to_gw):
    """Aggregate per-player stats across GW range with feature engineering."""
    if self.season == '2022-23' and from_gw == 7:
        from_gw = 8

    # Load data for first GW
    gk_data = self.get_pos_data(season, from_gw, 'GK')
    def_data = self.get_pos_data(season, from_gw, 'DEF')
    mid_data = self.get_pos_data(season, from_gw, 'MID')
    fwd_data = self.get_pos_data(season, from_gw, 'FWD')

    # Aggregate across GW range
    for i in range(from_gw + 1, to_gw + 1):
        if self.season == '2022-23' and i == 7:
            continue
        gk_data = pd.concat((gk_data, self.get_pos_data(season, i, 'GK')))
        def_data = pd.concat((def_data, self.get_pos_data(season, i, 'DEF')))
        mid_data = pd.concat((mid_data, self.get_pos_data(season, i, 'MID')))
        fwd_data = pd.concat((fwd_data, self.get_pos_data(season, i, 'FWD')))

    # === NEW: Apply feature engineering per position ===
    gk_data = engineer_features_on_gw_data(gk_data, season, to_gw, self, 'GK')
    def_data = engineer_features_on_gw_data(def_data, season, to_gw, self, 'DEF')
    mid_data = engineer_features_on_gw_data(mid_data, season, to_gw, self, 'MID')
    fwd_data = engineer_features_on_gw_data(fwd_data, season, to_gw, self, 'FWD')

    # Aggregate by player name (existing logic)
    gk_data = gk_data.groupby('name').mean().reset_index().set_index('name')
    def_data = def_data.groupby('name').mean().reset_index().set_index('name')
    mid_data = mid_data.groupby('name').mean().reset_index().set_index('name')
    fwd_data = fwd_data.groupby('name').mean().reset_index().set_index('name')

    return gk_data, def_data, mid_data, fwd_data
```

**Step 1: Extract RMSE per position per season from logs**

Write a Python script to parse logs:

```python
import json
import re

baseline_metrics_v3 = {}  # Phase 3 (existing)
v1_metrics = {}           # Phase 4 v1 (after feature engineering)

# Load Phase 3 baseline from BASELINE_METRICS.json (created in Phase 3)
with open('.planning/BASELINE_METRICS.json', 'r') as f:
    baseline_metrics_v3 = json.load(f)

# Parse Phase 4 v1 logs
for season in ['2021-22', '2022-23', '2023-24', '2024-25']:
    log_file = f'logs/phase4_v1_{season}.log'
    with open(log_file, 'r') as f:
        content = f.read()
    
    # Extract "Average Test RMSE:  GK: X.XX, DEF: Y.YY, MID: Z.ZZ, FWD: W.WW"
    match = re.search(r'Average Test RMSE:\s+GK: ([\d.]+), DEF: ([\d.]+), MID: ([\d.]+), FWD: ([\d.]+)', content)
    if match:
        v1_metrics[season] = {
            'gk_rmse': float(match.group(1)),
            'def_rmse': float(match.group(2)),
            'mid_rmse': float(match.group(3)),
            'fwd_rmse': float(match.group(4))
        }

# Calculate RMSE delta (improvement %)
results = {}
for season in ['2021-22', '2022-23', '2023-24', '2024-25']:
    baseline = baseline_metrics_v3[season]
    v1 = v1_metrics[season]
    
    gk_delta = (baseline['gk_rmse'] - v1['gk_rmse']) / baseline['gk_rmse'] * 100
    def_delta = (baseline['def_rmse'] - v1['def_rmse']) / baseline['def_rmse'] * 100
    mid_delta = (baseline['mid_rmse'] - v1['mid_rmse']) / baseline['mid_rmse'] * 100
    fwd_delta = (baseline['fwd_rmse'] - v1['fwd_rmse']) / baseline['fwd_rmse'] * 100
    
    results[season] = {
        'gk_delta_pct': round(gk_delta, 2),
        'def_delta_pct': round(def_delta, 2),
        'mid_delta_pct': round(mid_delta, 2),
        'fwd_delta_pct': round(fwd_delta, 2),
        'avg_delta_pct': round((gk_delta + def_delta + mid_delta + fwd_delta) / 4, 2)
    }

# Calculate overall 3-season average improvement
overall_avg = sum(r['avg_delta_pct'] for r in results.values()) / len(results)

print(f"Overall improvement: {overall_avg:.2f}%")
print(f"Target: >= 5.0%")
print(f"Status: {'PASS' if overall_avg >= 5.0 else 'FAIL'}")
```
